cmake_minimum_required(VERSION 3.0.2)
project(ebs_test)
find_package(catkin REQUIRED)
catkin_package()
catkin_install_python(PROGRAMS scripts/ebs_controller.py scripts/replay_diagnostics.py DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
configure_file(scripts/ebs_core.py ${CATKIN_DEVEL_PREFIX}/${CATKIN_PACKAGE_BIN_DESTINATION}/ebs_core.py COPYONLY)
configure_file(scripts/ebs_output.py ${CATKIN_DEVEL_PREFIX}/${CATKIN_PACKAGE_BIN_DESTINATION}/ebs_output.py COPYONLY)
install(FILES scripts/ebs_core.py scripts/ebs_output.py DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(DIRECTORY launch config DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
install(FILES README.md DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
